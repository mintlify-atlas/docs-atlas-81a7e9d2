## Overview

The SelectMenu component combines a select dropdown with search functionality, providing filtering, custom item creation, multiple selection, and virtualization for large lists.

## Basic Usage

```vue
<script setup lang="ts">
import { ref } from 'vue'

const selected = ref(null)
const items = [
  'Apple',
  'Banana',
  'Cherry',
  'Date',
  'Elderberry'
]
</script>

<template>
  <USelectMenu v-model="selected" :items="items" placeholder="Select a fruit" />
</template>
```

## With Objects

```vue
<script setup lang="ts">
import { ref } from 'vue'

const selected = ref(null)
```

## With Search Input

```vue
<template>
  <!-- Default search -->
  <USelectMenu v-model="selected" :items="items" search-input />
  
  <!-- Custom search placeholder -->
  <USelectMenu
    v-model="selected"
    :items="items"
    :search-input="{ placeholder: 'Search people...' }"
  />
  
  <!-- No search -->
  <USelectMenu v-model="selected" :items="items" :search-input="false" />
</template>
```

## Create Custom Items

```vue
<script setup lang="ts">
import { ref } from 'vue'

const selected = ref(null)
const items = ref(['Apple', 'Banana', 'Cherry'])

function onCreate(item: string) {
  items.value.push(item)
  selected.value = item
}
</script>

<template>
  <USelectMenu
    v-model="selected"
    :items="items"
    create-item
    @create="onCreate"
    placeholder="Type to create..."
  />
</template>
```

## Custom Filtering

```vue
<script setup lang="ts">
import { ref } from 'vue'

const selected = ref(null)
const searchTerm = ref('')
const items = ref([])

// Fetch items from API based on search
watchDebounced(searchTerm, async (value) => {
  if (!value) return
  const { data } = await useFetch('/api/search', {
    query: { q: value }
  })
  items.value = data.value
}, { debounce: 300 })
</script>

<template>
  <USelectMenu
    v-model="selected"
    v-model:search-term="searchTerm"
    :items="items"
    ignore-filter
    placeholder="Search..."
  />
</template>
```

## Virtualization

Efficiently render large lists:

```vue
<script setup lang="ts">
const items = Array.from({ length: 10000 }, (_, i) => ({
  label: `Item ${i + 1}`,
  value: i
}))
</script>

<template>
  <USelectMenu
    v-model="selected"
    :items="items"
    virtualize
    placeholder="Select from 10k items"
  />
</template>
```

## With Form

```vue
<script setup lang="ts">
import { z } from 'zod'
import { ref } from 'vue'

const schema = z.object({
  country: z.string().min(1, 'Please select a country'),
  skills: z.array(z.string()).min(1, 'Select at least one skill')
})

const state = ref({
  country: '',
  skills: []
})

const countries = [
  { value: 'us', label: 'United States' },
  { value: 'ca', label: 'Canada' },
  { value: 'mx', label: 'Mexico' }
]

const skills = [
  'JavaScript', 'TypeScript', 'Vue', 'React', 'Node.js'
]
</script>

<template>
  <UForm :schema="schema" :state="state">
    <UFormField label="Country" name="country">
      <USelectMenu v-model="state.country" :items="countries" />
    </UFormField>
    
    <UFormField label="Skills" name="skills">
      <USelectMenu v-model="state.skills" :items="skills" multiple />
    </UFormField>
  </UForm>
</template>
```

## Props

<ParamField path="modelValue" type="any">
  The controlled value. Can be bound with v-model.
</ParamField>

<ParamField path="defaultValue" type="any">
  The default value when initially rendered.
</ParamField>

<ParamField path="items" type="array">
  Array of items or nested arrays for grouped items.
</ParamField>

<ParamField path="valueKey" type="string">
  Field to use as the value when items are objects.
</ParamField>

<ParamField path="labelKey" type="string" default="label">
  Field to use as the label when items are objects.
</ParamField>

<ParamField path="descriptionKey" type="string" default="description">
  Field to use as the description when items are objects.
</ParamField>

<ParamField path="placeholder" type="string">
  Placeholder text when no item is selected.
</ParamField>

<ParamField path="searchInput" type="boolean | object" default="true">
  Display the search input. Can be an object to pass additional props.
</ParamField>

<ParamField path="multiple" type="boolean">
  Allow multiple items to be selected.
</ParamField>

<ParamField path="createItem" type="boolean | 'always' | object">
  Allow creating custom items.
</ParamField>

<ParamField path="filterFields" type="string[]">
  Fields to filter items by.
</ParamField>

<ParamField path="ignoreFilter" type="boolean">
  Disable default filtering.
</ParamField>

<ParamField path="virtualize" type="boolean | object">
  Enable virtualization for large lists.
</ParamField>

<ParamField path="searchTerm" type="string">
  The current search term. Can be bound with v-model:search-term.
</ParamField>

<ParamField path="clear" type="boolean | object">
  Show a clear button.
</ParamField>

<ParamField path="clearIcon" type="string">
  Icon for the clear button.
</ParamField>

<ParamField path="selectedIcon" type="string">
  Icon shown for selected items.
</ParamField>

<ParamField path="trailingIcon" type="string">
  Icon displayed at the end.
</ParamField>

<ParamField path="color" type="string" default="primary">
  Color variant.
</ParamField>

<ParamField path="variant" type="string" default="outline">
  Visual variant.
</ParamField>

<ParamField path="size" type="string" default="md">
  Size variant.
</ParamField>

<ParamField path="disabled" type="boolean">
  Disable the select.
</ParamField>

<ParamField path="portal" type="boolean | string | HTMLElement" default="true">
  Render the menu in a portal.
</ParamField>

<ParamField path="content" type="object">
  Content props (side, sideOffset, etc.).
</ParamField>

<ParamField path="arrow" type="boolean | object">
  Display an arrow alongside the menu.
</ParamField>

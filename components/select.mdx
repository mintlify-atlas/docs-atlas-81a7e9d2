## Overview

The Select component provides a customizable dropdown menu for selecting one or multiple items from a list, with support for icons, avatars, descriptions, and grouping.

## Basic Usage

```vue
<script setup lang="ts">
import { ref } from 'vue'

const selected = ref(null)
const items = [
  'Apple',
  'Banana',
  'Cherry',
  'Date',
  'Elderberry'
]
</script>

<template>
  <USelect v-model="selected" :items="items" placeholder="Select a fruit" />
</template>
```

## With Form

```vue
<script setup lang="ts">
import { z } from 'zod'
import { ref } from 'vue'

const schema = z.object({
  country: z.string().min(1, 'Please select a country'),
  role: z.string().min(1, 'Please select a role')
})

const state = ref({
  country: '',
  role: ''
})

const countries = [
  { value: 'us', label: 'United States' },
  { value: 'ca', label: 'Canada' },
  { value: 'mx', label: 'Mexico' }
]

const roles = [
  { value: 'admin', label: 'Admin' },
  { value: 'user', label: 'User' },
  { value: 'guest', label: 'Guest' }
]
</script>

<template>
  <UForm :schema="schema" :state="state">
    <UFormField label="Country" name="country">
      <USelect v-model="state.country" :items="countries" />
    </UFormField>
    
    <UFormField label="Role" name="role">
      <USelect v-model="state.role" :items="roles" />
    </UFormField>
  </UForm>
</template>
```

## Props

<ParamField path="modelValue" type="any">
  The controlled value. Can be bound with v-model.
</ParamField>

<ParamField path="defaultValue" type="any">
  The default value when initially rendered.
</ParamField>

<ParamField path="items" type="array">
  Array of items or nested arrays for grouped items.
</ParamField>

<ParamField path="valueKey" type="string" default="value">
  Field to use as the value when items are objects.
</ParamField>

<ParamField path="labelKey" type="string" default="label">
  Field to use as the label when items are objects.
</ParamField>

<ParamField path="descriptionKey" type="string" default="description">
  Field to use as the description when items are objects.
</ParamField>

<ParamField path="placeholder" type="string">
  Placeholder text when no item is selected.
</ParamField>

<ParamField path="multiple" type="boolean">
  Allow multiple items to be selected.
</ParamField>

<ParamField path="color" type="string" default="primary">
  Color variant.
</ParamField>

<ParamField path="variant" type="string" default="outline">
  Visual variant.
</ParamField>

<ParamField path="size" type="string" default="md">
  Size variant.
</ParamField>

<ParamField path="leadingIcon" type="string">
  Icon displayed at the start.
</ParamField>

<ParamField path="trailingIcon" type="string">
  Icon displayed at the end (chevron by default).
</ParamField>

<ParamField path="selectedIcon" type="string">
  Icon shown for selected items.
</ParamField>

<ParamField path="disabled" type="boolean">
  Disable the select.
</ParamField>

<ParamField path="required" type="boolean">
  Mark as required.
</ParamField>

<ParamField path="highlight" type="boolean">
  Highlight the ring color.
</ParamField>

<ParamField path="autofocus" type="boolean">
  Automatically focus on mount.
</ParamField>

<ParamField path="autofocusDelay" type="number" default="0">
  Delay before autofocus.
</ParamField>

<ParamField path="portal" type="boolean | string | HTMLElement" default="true">
  Render the menu in a portal.
</ParamField>

<ParamField path="content" type="object">
  Content props (side, sideOffset, etc.).
</ParamField>

<ParamField path="arrow" type="boolean | object">
  Display an arrow alongside the menu.
</ParamField>

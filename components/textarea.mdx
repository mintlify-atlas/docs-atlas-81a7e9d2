## Overview

The Textarea component provides a multi-line text input with support for auto-resizing, leading/trailing icons, different variants, and seamless integration with Form components.

## Basic Usage

```vue
<script setup lang="ts">
import { ref } from 'vue'

const message = ref('')
</script>

<template>
  <UTextarea v-model="message" placeholder="Enter your message..." />
</template>
```

## Auto-resize

```vue
<template>
  <!-- Auto-resize based on content -->
  <UTextarea
    v-model="message"
    autoresize
    placeholder="This will grow as you type"
  />
  
  <!-- With max rows -->
  <UTextarea
    v-model="message"
    autoresize
    :rows="3"
    :maxrows="10"
    placeholder="Grows up to 10 rows"
  />
</template>
```

## With Form

```vue
<script setup lang="ts">
import { z } from 'zod'
import { ref } from 'vue'

const schema = z.object({
  bio: z.string().min(10, 'Bio must be at least 10 characters'),
  notes: z.string().max(500, 'Notes cannot exceed 500 characters')
})

const state = ref({
  bio: '',
  notes: ''
})
</script>

<template>
  <UForm :schema="schema" :state="state">
    <UFormField label="Bio" name="bio">
      <UTextarea
        v-model="state.bio"
        autoresize
        placeholder="Tell us about yourself"
      />
    </UFormField>
    
    <UFormField label="Notes" name="notes">
      <UTextarea
        v-model="state.notes"
        :rows="5"
        placeholder="Additional notes"
      />
    </UFormField>
  </UForm>
</template>
```

## Character Count

```vue
<script setup lang="ts">
import { ref, computed } from 'vue'

const message = ref('')
const maxLength = 280
const remaining = computed(() => maxLength - message.value.length)
</script>

<template>
  <UFormField label="Message" :hint="`${remaining} characters remaining`">
    <UTextarea
      v-model="message"
      :maxlength="maxLength"
      placeholder="What's on your mind?"
    />
  </UFormField>
</template>
```

## Props

<ParamField path="modelValue" type="string | number | null">
  The controlled value. Can be bound with v-model.
</ParamField>

<ParamField path="defaultValue" type="string | number | null">
  The default value when initially rendered.
</ParamField>

<ParamField path="placeholder" type="string">
  Placeholder text when the textarea is empty.
</ParamField>

<ParamField path="name" type="string">
  The name attribute for the textarea.
</ParamField>

<ParamField path="id" type="string">
  The id attribute for the textarea.
</ParamField>

<ParamField path="rows" type="number" default="3">
  Number of visible text rows.
</ParamField>

<ParamField path="maxrows" type="number" default="0">
  Maximum number of rows when autoresize is enabled (0 = unlimited).
</ParamField>

<ParamField path="autoresize" type="boolean">
  Enable automatic height adjustment based on content.
</ParamField>

<ParamField path="autoresizeDelay" type="number" default="0">
  Delay in milliseconds before initial auto-resize.
</ParamField>

<ParamField path="color" type="'primary' | 'secondary' | 'success' | 'warning' | 'error' | 'neutral'" default="primary">
  Color variant of the textarea.
</ParamField>

<ParamField path="variant" type="'outline' | 'soft' | 'ghost' | 'none'" default="outline">
  Visual variant of the textarea.
</ParamField>

<ParamField path="size" type="'xs' | 'sm' | 'md' | 'lg' | 'xl'" default="md">
  Size variant of the textarea.
</ParamField>

<ParamField path="leadingIcon" type="string">
  Icon name to display at the start.
</ParamField>

<ParamField path="trailingIcon" type="string">
  Icon name to display at the end.
</ParamField>

<ParamField path="avatar" type="AvatarProps">
  Avatar configuration to display at the start.
</ParamField>

<ParamField path="loading" type="boolean">
  Show loading indicator.

<ParamField path="required" type="boolean">

<ParamField path="fixed" type="boolean">

<ParamField path="autofocus" type="boolean">

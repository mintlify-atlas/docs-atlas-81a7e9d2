## Usage

The CheckboxGroup component allows users to select multiple values from a set of options.

```vue
<template>
  <UCheckboxGroup
    v-model="selected"
    legend="Select your interests"
    :items="items"
  />
</template>

<script setup>
const selected = ref([])
const items = [
  { value: 'design', label: 'Design' },
  { value: 'development', label: 'Development' },
  { value: 'marketing', label: 'Marketing' }
]
</script>
```

## Props

<ParamField path="modelValue" type="array">
  The controlled value of the checkbox group. Can be bound with `v-model`.
</ParamField>

<ParamField path="defaultValue" type="array">
  The value of the checkbox group when initially rendered.
</ParamField>

<ParamField path="items" type="array">
  Array of items to render as checkboxes. Can be strings, numbers, or objects.
</ParamField>

<ParamField path="legend" type="string">
  The legend text for the checkbox group fieldset.
</ParamField>

<ParamField path="valueKey" type="string" default="value">
  When items is an array of objects, select the field to use as the value.
</ParamField>

<ParamField path="labelKey" type="string" default="label">
  When items is an array of objects, select the field to use as the label.
</ParamField>

<ParamField path="descriptionKey" type="string" default="description">
  When items is an array of objects, select the field to use as the description.
</ParamField>

<ParamField path="size" type="string" default="md">
  The size of the checkboxes. Options: `sm`, `md`, `lg`.
</ParamField>

<ParamField path="variant" type="string" default="list">
  The visual variant of the checkboxes.
</ParamField>

<ParamField path="color" type="string">
  The color theme of the checkboxes.
</ParamField>

<ParamField path="orientation" type="string" default="vertical">
  The orientation the checkboxes are laid out. Options: `vertical`, `horizontal`.
</ParamField>

<ParamField path="indicator" type="string" default="start">
  Position of the checkbox indicator. Options: `start`, `end`.
</ParamField>

<ParamField path="icon" type="string">
  The icon displayed when checked.
</ParamField>

<ParamField path="disabled" type="boolean">
  When `true`, prevents user interaction with all checkboxes.
</ParamField>

<ParamField path="required" type="boolean">
  When `true`, indicates that at least one checkbox must be checked.
</ParamField>

<ParamField path="name" type="string">
  The name of the checkbox group for form submission.
</ParamField>

<ParamField path="loop" type="boolean">
  When `true`, keyboard navigation will loop from last to first item.
</ParamField>

<ParamField path="as" type="any" default="div">
  The element or component this component should render as.
</ParamField>

<ParamField path="class" type="any">
  Additional CSS classes to apply to the root element.
</ParamField>

<ParamField path="ui" type="object">
  UI customization object for styling different parts of the checkbox group.
</ParamField>

## Events

<ParamField path="@change" type="(event: Event) => void">
  Emitted when any checkbox in the group changes.
</ParamField>

<ParamField path="@update:modelValue" type="(value: array) => void">
  Emitted when the selected values change.
</ParamField>
